// Inicializa el cliente de Supabase con tus credenciales públicas
const SUPABASE_URL = 'https://iqndiyuhcjxrugzfyjaj.supabase.co'; // Tu URL de Supabase
const SUPABASE_ANON_KEY = 'TU_ANON_KEY_PUBLICA'; // Tu llave anónima (anon public) de Supabase

const supabaseClient = supabase.createClient(SUPABASE_URL, SUPABASE_ANON_KEY);

async function cargarJuegos() {
  const grid = document.getElementById('games-grid');
  
  // Consultamos la tabla games y la tabla game_prices
  const { data: games, error } = await supabaseClient
    .from('games')
    .select('*, game_prices(*)');

  if (error) {
    console.error('Error al cargar juegos:', error);
    grid.innerHTML = '<p>Error al cargar el catálogo.</p>';
    return;
  }

  if (!games || games.length === 0) {
    grid.innerHTML = '<p>No hay juegos cargados todavía.</p>';
    return;
  }

  // Dibujamos los juegos en la pantalla
  grid.innerHTML = games.map(game => {
    // Buscamos el precio más bajo o mostramos las ofertas
    const precios = game.game_prices || [];
    const mejorPrecio = precios.length > 0 ? Math.min(...precios.map(p => p.price)) : 'Próximamente';

    return `
      <div class="game-card">
        <img src="${game.cover_image || ''}" alt="${game.title}" onerror="this.src='https://via.placeholder.com/300x400?text=Sin+Imagen'">
        <div class="game-info">
          <h3>${game.title}</h3>
          <p class="platform">${game.platform}</p>
          <p class="price">Desde $${mejorPrecio}</p>
        </div>
      </div>
    `;
  }).join('');
}

// Ejecutar al cargar la página
document.addEventListener('DOMContentLoaded', cargarJuegos);
